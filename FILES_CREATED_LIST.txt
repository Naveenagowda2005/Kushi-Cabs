     Condition: bucket_id = 'odometer-images'

  2. Policy SELECT (Public Read)
     Role: public
     Condition: bucket_id = 'odometer-images'

  3. Policy SELECT (Authenticated Read)
     Role: authenticated
     Condition: bucket_id = 'odometer-images'

  4. Policy DELETE (Own Images)
     Role: authenticated
     Condition: bucket_id = 'odometer-images' AND owner_id = auth.uid()
